armor_super_min_production_ = {
	enable = {
		num_of_military_factories > 150
		ai_has_role_template = armour_super_heavy
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = equipment_production_min_factories_archetype 	id = heavy_tank_aa_chassis	 		value = 3 }

}
